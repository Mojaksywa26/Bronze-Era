# Mycenae Outcomes
trojan_war_raid_on_anatolia_success.1 = {
    type = country_event
    title = trojan_war_raid_on_anatolia_success.t
    desc = trojan_war_raid_on_anatolia_success.d
    
     trigger = {
         NOT = {
            has_global_variable = trojan_war_phase_1_ended
         }
    }
    
    hidden = no 

	 immediate = {
        if = {
            limit = {
                NOT = { has_global_variable = anatolia_raid_success_count }
            }
            set_global_variable = {
                name = anatolia_raid_success_count
                value = 0
            }
        }
        change_global_variable = { 
            name = anatolia_raid_success_count 
            add = 1 
        }
    #  if = {
    #    limit = {
    #        global_var:anatolia_raid_success_count = 1
    #    }
    #    error_log = "Anatolia raid success count works"
    # }

    }

    option = {
        name = trojan_war_raid_on_anatolia_success.a
        add_gold = 100
		add_prestige = 20                         
        custom_flavor_text = "treasure_secured" 
    }
	option = {
		name = trojan_war_raid_on_anatolia_success.b
		add_gold = 150
		custom_flavor_text = "treasure_secured"
	}
}
